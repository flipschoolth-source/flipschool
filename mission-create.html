async function callGeminiAPI(apiKey, prompt) { // ตรวจสอบว่ามี API Key หรือไม่ if (!apiKey) return { success: false, error: "ไม่พบ API Key" }; // URL สำหรับเรียกใช้งาน Gemini 1.5 Flash const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`; try { const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: { temperature: 0.7, maxOutputTokens: 4000 } }) }); const data = await response.json(); if (!response.ok) { // วิเคราะห์ Error ตามรหัสที่ Google ส่งกลับมา console.error("API Error Data:", data); let errorMsg = data.error?.message || "เกิดข้อผิดพลาดไม่ทราบสาเหตุ"; if (response.status === 400) errorMsg = "คำขอผิดพลาด (Bad Request) - ตรวจสอบรูปแบบ API Key"; if (response.status === 403) errorMsg = "สิทธิ์การใช้งานถูกปฏิเสธ - API Key อาจยังไม่เปิดใช้งาน Generative Language API"; if (response.status === 404) errorMsg = "ไม่พบโมเดล - ตรวจสอบชื่อรุ่น gemini-1.5-flash ในโค้ด"; if (response.status === 429) errorMsg = "เรียกใช้งานบ่อยเกินไป (Quota Exceeded) - กรุณารอ 1 นาที"; return { success: false, error: errorMsg }; } return { success: true, content: data.candidates[0].content.parts[0].text }; } catch (error) { console.error("Fetch Error:", error); return { success: false, error: "ไม่สามารถเชื่อมต่อกับเซิร์ฟเวอร์ Google ได้ (ตรวจสอบอินเทอร์เน็ต)" }; } }
